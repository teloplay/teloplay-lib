import 'dart:async';
import 'dart:js_interop';

import 'package:web/web.dart' as web;

import '../logging/app_logger.dart';

/// Browser-native HTMLAudioElement wrapper.
///
/// iOS Safari background + lock-screen Media Session only attach to a real
/// HTML <audio> element, not CanvasKit/mpv. This stays app-controlled so
/// later Supabase/history/queue wiring can sit on top of the same API.
class WebHtmlAudioPlayer {
  WebHtmlAudioPlayer() {
    _audio = web.HTMLAudioElement()
      ..preload = 'auto'
      ..controls = false
      ..setAttribute('playsinline', 'true')
      ..setAttribute('webkit-playsinline', 'true');
    web.document.body?.append(_audio);

    _playingSub = _audio.onPlay.listen((_) {
      _playing = true;
      _playingController.add(true);
    });
    _pauseSub = _audio.onPause.listen((_) {
      _playing = false;
      _playingController.add(false);
    });
    _timeSub = _audio.onTimeUpdate.listen((_) {
      _positionController.add(Duration(
        milliseconds: (_audio.currentTime * 1000).round(),
      ));
    });
    _metaSub = _audio.onDurationChange.listen((_) {
      final d = _audio.duration;
      if (d.isFinite && d > 0) {
        _durationController.add(Duration(milliseconds: (d * 1000).round()));
      }
    });
    _waitSub = _audio.onWaiting.listen((_) => _bufferingController.add(true));
    _canPlaySub =
        _audio.onCanPlay.listen((_) => _bufferingController.add(false));
    _errorSub = _audio.onError.listen((_) {
      final err = _audio.error;
      final code = err?.code;
      final meaning = switch (code) {
        web.MediaError.MEDIA_ERR_ABORTED => 'aborted',
        web.MediaError.MEDIA_ERR_NETWORK => 'network failure while fetching',
        web.MediaError.MEDIA_ERR_DECODE => 'decode failure',
        web.MediaError.MEDIA_ERR_SRC_NOT_SUPPORTED => 'source not supported',
        _ => 'unknown',
      };
      AppLogger.playback(
        '[web-audio] element error code=$code ($meaning)'
        '${err == null || err.message.isEmpty ? '' : ' msg=${err.message}'}',
      );
      _errorController.add('code=$code ($meaning)');
    });
    // Apply a deferred seek once metadata is ready. Used by the recovery flow
    // in main_web.dart so a recovered stream resumes near the old position.
    _loadedMetaSub = _audio.onLoadedMetadata.listen((_) {
      final target = _pendingSeek;
      if (target != null && target > Duration.zero) {
        _pendingSeek = null;
        try {
          _audio.currentTime = target.inMilliseconds / 1000.0;
          AppLogger.playback(
              '[web-audio] resumed at ${target.inSeconds}s after recovery');
        } catch (_) {}
      }
    });
    _endedSub = _audio.onEnded.listen((_) {
      _playing = false;
      _playingController.add(false);
      _completedController.add(true);
    });
  }

  late final web.HTMLAudioElement _audio;

  final _playingController = StreamController<bool>.broadcast();
  final _positionController = StreamController<Duration>.broadcast();
  final _durationController = StreamController<Duration?>.broadcast();
  final _bufferingController = StreamController<bool>.broadcast();
  final _completedController = StreamController<bool>.broadcast();
  /// Human-readable description of the last media element error
  /// (`code=N (meaning)`), emitted on each element error event.
  final _errorController = StreamController<String>.broadcast();

  StreamSubscription<web.Event>? _playingSub;
  StreamSubscription<web.Event>? _pauseSub;
  StreamSubscription<web.Event>? _timeSub;
  StreamSubscription<web.Event>? _metaSub;
  StreamSubscription<web.Event>? _waitSub;
  StreamSubscription<web.Event>? _canPlaySub;
  StreamSubscription<web.Event>? _errorSub;
  StreamSubscription<web.Event>? _loadedMetaSub;
  StreamSubscription<web.Event>? _endedSub;

  bool _playing = false;
  /// Seek target to apply once the next `loadedmetadata` fires. Set by
  /// [open] so recovery can resume playback near the previous position.
  Duration? _pendingSeek;
  bool get playing => _playing;

  Stream<bool> get playingStream => _playingController.stream;
  Stream<Duration> get positionStream => _positionController.stream;
  Stream<Duration?> get durationStream => _durationController.stream;
  Stream<bool> get bufferingStream => _bufferingController.stream;
  Stream<bool> get completedStream => _completedController.stream;
  Stream<String> get errorStream => _errorController.stream;

  /// Opens [url] and, once metadata loads, seeks to [resumeAt] before
  /// playback starts. Used for mid-track recovery so the listener keeps
  /// their place instead of restarting the song from zero.
  Future<void> open(String url, {Duration? resumeAt}) async {
    _pendingSeek = resumeAt ?? Duration.zero;
    // Assigning a fresh src resets the media element error state; load()
    // then kicks off the new (recovered) fetch.
    _audio.src = url;
    _audio.load();
  }

  Future<void> play() async {
    try {
      await _audio.play().toDart;
    } catch (e) {
      AppLogger.playback('[web-audio] play() blocked or failed: $e');
      rethrow;
    }
  }

  Future<void> pause() async {
    _audio.pause();
  }

  Future<void> stop() async {
    _pendingSeek = null;
    _audio.pause();
    _audio.removeAttribute('src');
    _audio.load();
    _playing = false;
    _playingController.add(false);
  }

  Future<void> seek(Duration position) async {
    _audio.currentTime = position.inMilliseconds / 1000.0;
  }

  Future<void> dispose() async {
    await stop();
    await _playingSub?.cancel();
    await _pauseSub?.cancel();
    await _timeSub?.cancel();
    await _metaSub?.cancel();
    await _waitSub?.cancel();
    await _canPlaySub?.cancel();
    await _errorSub?.cancel();
    await _loadedMetaSub?.cancel();
    await _endedSub?.cancel();
    _audio.remove();
    await _playingController.close();
    await _positionController.close();
    await _durationController.close();
    await _bufferingController.close();
    await _completedController.close();
    await _errorController.close();
  }
}
